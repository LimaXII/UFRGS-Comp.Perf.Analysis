\begin{frame}{Introdução e Objetivos}
    \textbf{O Desafio:}
    \begin{itemize}

    \textbf{Objetivo do Estudo:}

    \begin{itemize}
        \item Criação de 10 documentos de referência formatados em Markdown, originalmente em Inglês (\texttt{en\_us}).
    \end{itemize}
    
    \vspace{0.3cm}
    \textbf{Processo de Tradução:}
    \begin{itemize}
        \item Expansão linguística conduzida por um LLM operando em modo \textit{zero-shot}.
        \item Aplicação de restrições no prompt para garantir a preservação estrita da formatação.
        \item Prevenção contra sumarizações automáticas ou perda de numeração de linhas, garantindo paridade semântica.
    \end{itemize}
\end{frame}

    \begin{table}
        \centering
        \small
        \begin{tabular}{ll}
        \hline
        \textbf{Variável Controlada} & \textbf{Níveis} \\ \hline
        Tamanho do Lote (\textit{Batch Size}) & 7 configurações \\

        Precisão Numérica (\textit{Precision}) & \texttt{float32} e \texttt{float16} \\
        Métrica de Distância (FAISS) & L2 e \textit{Inner Product} \\ \hline
        \end{tabular}
    \end{table}

    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Cobertura:} 20 idiomas avaliados em 10 replicações.
        \item \textbf{Volume de Dados:} 39.200 configurações totais durante a geração das bases de dados.
        \item \textbf{Processamento Exclusivo (CUDA):} A variável \textit{device} (CPU vs. GPU) foi removida do escopo. A execução combinatória do Fatorial Completo em CPU tornaria o tempo de experimento impraticável.
    \end{itemize}
\end{frame}

% --- 2.8 ESTRUTURA DAS CONSULTAS ---
\begin{frame}{2.8 Estrutura das Consultas (\textit{Queries})}
    \textbf{Elaboração e Mapeamento \textit{Gold Standard}:}
    \begin{itemize}
        \item Foram criadas 20 perguntas, traduzidas seguindo a mesma metodologia do corpus de base.
        \item Cada documento da base contém a resposta direta para exatamente 2 dessas perguntas.
        \item Isso estabelece um \textit{Gold Standard} determinístico: para cada pergunta, existe \textbf{uma única resposta certa} a ser encontrada no índice.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Expansão Volumétrica dos Dados:}
    \begin{itemize}
        \item A matriz do DoE resultou em 39.200 bancos de dados gerados.
        \item Ao cruzar esse cenário com as 20 consultas, o arquivo \textit{CSV} resultante atingiu a marca de \textbf{784.000 instâncias de avaliação}.
    \end{itemize}
\end{frame}

% --- 2.9 FOCO NO RETRIEVAL ---
\begin{frame}{2.9 Foco no \textit{Retrieval} e Custo Computacional}
    \textbf{Simplificação Arquitetural:}
    \begin{itemize}
        \item Inicialmente, havia um agente LLM projetado para formular a resposta final baseada nos documentos recuperados. Este passo foi removido. O escopo da pesquisa foca estritamente na qualidade do \textbf{modelo de \textit{embeddings}} e na busca semântica. Adicionar a geração final não traria novos dados métricos, apenas aumentaria o tempo de execução dos experimentos.
    \end{itemize}

    \vspace{0.3cm}
    \textbf{Avaliação Exclusiva e Tempo de Execução:}
    \begin{itemize}
        \item O pipeline avaliou exclusivamente a etapa de \textbf{\textit{Retrieval}} (capacidade de encontrar o documento correto no Top-K).
        \item Mesmo com essa otimização arquitetural, o processamento e a avaliação dessas 784.000 consultas exigiu cerca de \textbf{70 horas} de execução.
    \end{itemize}
\end{frame}

% --- DISCUSSÃO E CONCLUSÕES ---
\begin{frame}{Discussão e Conclusões}
    \textbf{Conclusões Principais:}
    \begin{enumerate}
        \item \textbf{A língua dita a eficiência:} A verbosidade afeta diretamente o armazenamento em disco e a latência de geração de \textit{embeddings}.
        \item \textbf{Viés do Modelo Base:} Observa-se uma disparidade de desempenho em favor do idioma inglês e francês, o que pode refletir de uma maior densidade de dados de treinamento nesses idiomas no modelo multilingue.
        \item \textbf{Falhas em Línguas Complexas:} Idiomas como \texttt{sv\_se} (Sueco) e \texttt{zh\_tw} (Chinês Tradicional) apresentaram falhas críticas na etapa de \textit{retrieval}.
    \end{enumerate}

    \vspace{0.3cm}
    \textbf{Trabalhos Futuros:}
    \begin{itemize}
        \item Investigar \textit{fine-tuning} para o modelo de \textit{embeddings} com o objetivo de melhorar o desempenho em idiomas de menor performance.
        \item Testar outro modelos de \textit{embeddings} multilingues para avaliar se o viés observado é específico do modelo atual ou uma tendência geral.
    \end{itemize}
\end{frame}
